vector<int> ZOV(int n, string s) {
    vector<int> z(n);
    int l = 0, r = 0;
    for (int i = 1; i < n; i++) {
        if (r > i) z[i] = min(r - i, z[i - l]);
        while (i + z[i] < n and s[z[i]] == s[i + z[i]]) z[i]++;
        if (z[i] + i > r) {
            r = i + z[i];
            l = i;
        }
    }
    return z;
}
